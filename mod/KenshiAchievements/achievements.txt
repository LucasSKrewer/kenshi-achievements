# Kenshi Achievements — edit freely, read when the game starts.
#
# Format:  id | metric | target | title | description
#
# Metrics:
#   kills, kos, takedowns (kills + KOs)            -> whole squad
#   char_kills, char_kos                           -> best single character
#   race_kills:<Race>, race_kos:<Race>             -> e.g. race_kills:Shek, race_kills:*Beak Thing
#   faction_kills:<Faction>, faction_kos:<Faction> -> e.g. faction_kills:*Holy Nation*
#
# Race/faction use the game's ENGLISH names, in any game language: translated names are mapped
# back to English using the game's own dictionary (locale/<lang>/gamedata.po). Case doesn't matter.
#   *   matches anything — useful with mods like Genesis that add variants ("*Skeleton*" = Skeleton MKI, MKII...)
#   ,   separates alternatives — "Dust Bandits,Hungry Bandits"
#
# Title/description here are the English default. Translations go in lang/<language>.txt
# (<id>.title / <id>.desc). Never change an `id` after release: saves track unlocks by id.

# Sound when an achievement unlocks:
#   @sound = Notifications:Building_Complete -> Kenshi's own sound (default), follows the game volume.
#       Other game notifications: Notifications:Thing_Complete, Notifications:Raid,
#       Notifications:Bounty_Fulfilled, Notifications:New_Biome
#       Other events: Character_Unlock_Success, Stats_Open, Tech_Open, Building_Confirm
#   @sound = achievement.wav                 -> any .wav in this folder (the bundled chime, or your own)
#   @sound = none                            -> silent
@sound = Notifications:Building_Complete

# Stats panel key: F1-F12, a letter/digit, or NUM0-NUM9.
# Won't open with Ctrl/Shift/Alt held or while typing in a text field.
# Avoid keys Kenshi already uses: F1-F5, F7-F9 (F9 = quickload), I, M, C, T, B, Y, R...
@key = F6

# Diagnostics: 1 logs every kill/KO/hit to RE_Kenshi_log.txt (keep 0 for normal play).
@debug = 0
